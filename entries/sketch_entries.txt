# classify_t: heavy-hitter detection based on frequency estimate
# Uses LPM on meta.freq_est (32-bit). Since CMS may over-count by a few
# percent, we match exact values that the demo is expected to produce.
# A production deployment would implement a proper threshold comparison.
table_add classify_t mark_heavy 100/32 =>    # freq_est = 100 (Flow A)
table_add classify_t mark_heavy 200/32 =>    # freq_est = 200 (Flow C)
table_add classify_t mark_normal 0/0 =>      # default: normal
